title: Headless React Hooks

---

The `@bolkauth/react` package exposes six type-safe headless hooks. "Headless" means they handle all auth logic and state, but ship **zero UI** — you bring your own components and styles.

---

## 📦 Installation & Setup

```bash
pnpm add @bolkauth/react
```

Wrap your application tree with `BolkAuthProvider` once — typically in `app/layout.tsx`:

```tsx
// app/layout.tsx
import { BolkAuthProvider } from "@bolkauth/react";

export default function RootLayout({ children }: { children: React.ReactNode }) {
  return (
    <html lang="en">
      <body>
        <BolkAuthProvider
          config={{
            baseURL: "/api/auth",   // must point to your bolkAuthHandler mount
          }}
        >
          {children}
        </BolkAuthProvider>
      </body>
    </html>
  );
}
```

### `BolkAuthProviderConfig`

```ts
interface BolkAuthProviderConfig {
  /** Base URL for the BolkAuth API. Default: "/api/auth" */
  baseURL?: string;

  /**
   * Callback fired when the user's auth state changes.
   * Useful for analytics or side effects.
   */
  onAuthStateChange?: (user: User | null) => void;
}
```

## 🔑 `useAuth()`

The lowest-level hook. Returns the raw authentication state from the provider context — no network requests made.

### Signature

```ts
function useAuth(): UseAuthReturn;

interface UseAuthReturn {
  /** True once the initial auth state has been loaded from the server */
  isLoaded: boolean;

  /** True if a valid session exists */
  isSignedIn: boolean;

  /** The current user's ID, or null if not signed in */
  userId: string | null;

  /** The current session ID, or null if not signed in */
  sessionId: string | null;
}
```

### Example — Conditional UI in a Header

```tsx
"use client";

import { useAuth } from "@bolkauth/react";

export function Header() {
  const { isLoaded, isSignedIn, userId } = useAuth();

  // Show a skeleton while auth state loads
  if (!isLoaded) {
    return (
      <header className="h-16 border-b flex items-center px-6">
        <div className="ml-auto w-20 h-8 bg-gray-200 rounded animate-pulse" />
      </header>
    );
  }

  return (
    <header className="h-16 border-b flex items-center justify-between px-6">
      <span className="font-bold text-lg">MyApp</span>
      {isSignedIn ? (
        <div className="flex items-center gap-4">
          <span className="text-sm text-gray-500">ID: {userId}</span>
          <a href="/dashboard" className="text-sm font-medium">Dashboard →</a>
        </div>
      ) : (
        <div className="flex items-center gap-3">
          <a href="/sign-in" className="text-sm">Sign In</a>
          <a href="/sign-up" className="text-sm bg-black text-white px-3 py-1.5 rounded">
            Get Started
          </a>
        </div>
      )}
    </header>
  );
}
```

## 👤 `useUser()`

Returns the full user object fetched from the database. Re-fetches whenever the auth state changes.

### Signature

```ts
function useUser(): UseUserReturn;

interface UseUserReturn {
  /** True once the user data has been fetched */
  isLoaded: boolean;

  /** True if a user is signed in */
  isSignedIn: boolean;

  /**
   * Full user object from the database.
   * null if not signed in or still loading.
   */
  user: User | null;
}

interface User {
  id: string;
  email: string;
  name: string | null;
  image: string | null;
  emailVerified: Date | null;
  createdAt: Date;
  metadata: UserMetadata | null;
}

interface UserMetadata {
  onboardingState: "NOT_STARTED" | "IN_PROGRESS" | "COMPLETE";
  onboardingData: Record<string, unknown> | null;
  customData: Record<string, unknown> | null;
}
```

### Example — User Avatar Component

```tsx
"use client";

import { useUser } from "@bolkauth/react";

export function UserAvatar() {
  const { isLoaded, isSignedIn, user } = useUser();

  if (!isLoaded) {
    return <div className="w-9 h-9 rounded-full bg-gray-200 animate-pulse" />;
  }

  if (!isSignedIn || !user) return null;

  const initials = user.name
    ? user.name.split(" ").map((n) => n[0]).join("").toUpperCase()
    : user.email[0].toUpperCase();

  return (
    <div className="flex items-center gap-3">
      {user.image ? (
        <img
          src={user.image}
          alt={user.name ?? "User"}
          className="w-9 h-9 rounded-full object-cover"
        />
      ) : (
        <div className="w-9 h-9 rounded-full bg-indigo-600 flex items-center justify-center text-white text-sm font-bold">
          {initials}
        </div>
      )}
      <div>
        <p className="text-sm font-medium leading-none">{user.name ?? user.email}</p>
        {user.name && (
          <p className="text-xs text-gray-500 mt-0.5">{user.email}</p>
        )}
      </div>
    </div>
  );
}
```

## 🔐 `useSignIn()`

Handles email/password sign-in with full loading and error state management.

### Signature

```ts
function useSignIn(): UseSignInReturn;

interface UseSignInReturn {
  /** Trigger a sign-in with email and password */
  signIn: (credentials: SignInCredentials) => Promise<SignInResult>;

  /** True while the sign-in network request is in-flight */
  isLoading: boolean;

  /** Set if the last sign-in attempt returned an error */
  error: AuthError | null;

  /** Reset error state (e.g., when the user starts typing again) */
  reset: () => void;
}

interface SignInCredentials {
  email: string;
  password: string;
}

interface SignInResult {
  user: User | null;
  error: AuthError | null;
}

interface AuthError {
  code: string;       // e.g. "INVALID_CREDENTIALS"
  message: string;    // human-readable
  status: number;     // HTTP status
}
```

### Example — Full Sign-In Form

```tsx
"use client";

import { useSignIn } from "@bolkauth/react";
import { useState } from "react";
import { useRouter } from "next/navigation";

export function SignInForm() {
  const router = useRouter();
  const { signIn, isLoading, error, reset } = useSignIn();
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    const result = await signIn({ email, password });

    if (!result.error) {
      router.push("/dashboard");
      router.refresh();
    }
  };

  return (
    <div className="max-w-sm mx-auto mt-16">
      <h1 className="text-2xl font-bold mb-6">Sign In</h1>

      {error && (
        <div className="mb-4 p-3 bg-red-50 border border-red-200 rounded-lg text-sm text-red-700">
          {error.code === "INVALID_CREDENTIALS"
            ? "Incorrect email or password. Please try again."
            : error.message}
        </div>
      )}

      <form onSubmit={handleSubmit} className="space-y-4">
        <div>
          <label className="block text-sm font-medium mb-1">Email</label>
          <input
            type="email"
            value={email}
            onChange={(e) => { setEmail(e.target.value); reset(); }}
            className="w-full border rounded-lg p-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-black"
            required
            autoComplete="email"
          />
        </div>

        <div>
          <label className="block text-sm font-medium mb-1">Password</label>
          <input
            type="password"
            value={password}
            onChange={(e) => { setPassword(e.target.value); reset(); }}
            className="w-full border rounded-lg p-2.5 text-sm focus:outline-none focus:ring-2 focus:ring-black"
            required
            autoComplete="current-password"
          />
        </div>

        <button
          type="submit"
          disabled={isLoading}
          className="w-full bg-black text-white py-2.5 rounded-lg text-sm font-medium hover:bg-gray-800 disabled:opacity-50 transition-colors"
        >
          {isLoading ? "Signing in..." : "Sign In"}
        </button>
      </form>

      <p className="mt-4 text-center text-sm text-gray-500">
        No account?{" "}
        <a href="/sign-up" className="text-black font-medium underline underline-offset-2">
          Sign up free
        </a>
      </p>
    </div>
  );
}
```

## ✏️ `useSignUp()`

Handles new user registration with email and password.

### Signature

```ts
function useSignUp(): UseSignUpReturn;

interface UseSignUpReturn {
  signUp: (data: SignUpData) => Promise<SignUpResult>;
  isLoading: boolean;
  error: AuthError | null;
  reset: () => void;
}

interface SignUpData {
  email: string;
  password: string;
  name?: string;
}

interface SignUpResult {
  user: User | null;
  error: AuthError | null;
  /** True if email verification is required before sign-in */
  requiresEmailVerification?: boolean;
}
```

### Example — Sign-Up Form with Password Strength Indicator

```tsx
"use client";

import { useSignUp } from "@bolkauth/react";
import { useState } from "react";
import { useRouter } from "next/navigation";

function getPasswordStrength(password: string): { score: number; label: string; color: string } {
  const checks = [
    password.length >= 8,
    /[A-Z]/.test(password),
    /[0-9]/.test(password),
    /[^A-Za-z0-9]/.test(password),
  ];
  const score = checks.filter(Boolean).length;
  const labels = ["", "Weak", "Fair", "Good", "Strong"];
  const colors = ["", "bg-red-400", "bg-yellow-400", "bg-blue-400", "bg-green-400"];
  return { score, label: labels[score] || "", color: colors[score] || "" };
}

export function SignUpForm() {
  const router = useRouter();
  const { signUp, isLoading, error, reset } = useSignUp();
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const strength = getPasswordStrength(password);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    const result = await signUp({ name, email, password });

    if (!result.error) {
      // Redirect to onboarding (or email verification page if configured)
      router.push(result.requiresEmailVerification ? "/verify-email" : "/onboarding");
    }
  };

  return (
    <div className="max-w-sm mx-auto mt-16">
      <h1 className="text-2xl font-bold mb-6">Create Your Account</h1>

      {error && (
        <div className="mb-4 p-3 bg-red-50 border border-red-200 rounded-lg text-sm text-red-700">
          {error.code === "EMAIL_ALREADY_EXISTS"
            ? "An account with this email already exists."
            : error.message}
        </div>
      )}

      <form onSubmit={handleSubmit} className="space-y-4">
        <div>
          <label className="block text-sm font-medium mb-1">Full Name</label>
          <input
            type="text"
            value={name}
            onChange={(e) => setName(e.target.value)}
            className="w-full border rounded-lg p-2.5 text-sm"
            placeholder="Jane Smith"
          />
        </div>

        <div>
          <label className="block text-sm font-medium mb-1">Email</label>
          <input
            type="email"
            value={email}
            onChange={(e) => { setEmail(e.target.value); reset(); }}
            className="w-full border rounded-lg p-2.5 text-sm"
            required
          />
        </div>

        <div>
          <label className="block text-sm font-medium mb-1">Password</label>
          <input
            type="password"
            value={password}
            onChange={(e) => { setPassword(e.target.value); reset(); }}
            className="w-full border rounded-lg p-2.5 text-sm"
            required
            minLength={8}
          />
          {password && (
            <div className="mt-2 space-y-1">
              <div className="flex gap-1">
                {[1, 2, 3, 4].map((i) => (
                  <div
                    key={i}
                    className={`h-1 flex-1 rounded-full transition-colors ${
                      i <= strength.score ? strength.color : "bg-gray-200"
                    }`}
                  />
                ))}
              </div>
              <p className="text-xs text-gray-500">
                Password strength: <span className="font-medium">{strength.label}</span>
              </p>
            </div>
          )}
        </div>

        <button
          type="submit"
          disabled={isLoading || strength.score < 2}
          className="w-full bg-black text-white py-2.5 rounded-lg text-sm font-medium disabled:opacity-50"
        >
          {isLoading ? "Creating account..." : "Create Account"}
        </button>
      </form>
    </div>
  );
}
```

## 🌐 `useOAuth()`

Initiates social OAuth flows. Redirects the browser to the provider's authorization URL.

### Signature

```ts
function useOAuth(): UseOAuthReturn;

interface UseOAuthReturn {
  /**
   * Redirects the user to the OAuth provider's authorization page.
   * The provider must be configured in createBolkAuth().
   */
  signInWithOAuth: (provider: OAuthProvider) => Promise<void>;

  /** True while the redirect URL is being fetched */
  isLoading: boolean;

  error: AuthError | null;
}

type OAuthProvider = "github" | "google" | "discord" | "twitter";
```

## 🧭 `useOnboarding()`

Manages multi-step onboarding flow state.

### Signature

```ts
function useOnboarding(): UseOnboardingReturn;

interface UseOnboardingReturn {
  /** Current onboarding state from user metadata */
  state: "NOT_STARTED" | "IN_PROGRESS" | "COMPLETE" | null;

  /** The accumulated data from all completed steps */
  data: Record<string, unknown> | null;

  /**
   * Persist data for the current step.
   * Data is merged with previous steps (not replaced).
   */
  saveStep: (stepData: Record<string, unknown>) => Promise<void>;

  /** Mark onboarding as complete */
  complete: () => Promise<void>;

  isLoading: boolean;
  error: AuthError | null;
}
```

### Example — Multi-Step Onboarding Wizard

```tsx
"use client";

import { useOnboarding } from "@bolkauth/react";
import { useState } from "react";
import { useRouter } from "next/navigation";

const STEPS = ["Welcome", "Your Role", "Your Team", "Done"] as const;

export function OnboardingWizard() {
  const router = useRouter();
  const { saveStep, complete, isLoading, error } = useOnboarding();
  const [currentStep, setCurrentStep] = useState(0);

  // Step 1 — Role
  const [role, setRole] = useState("");
  // Step 2 — Team
  const [teamSize, setTeamSize] = useState("");

  const handleRoleNext = async () => {
    await saveStep({ role });
    setCurrentStep(2);
  };

  const handleTeamNext = async () => {
    await saveStep({ teamSize });
    setCurrentStep(3);
  };

  const handleComplete = async () => {
    await complete();
    router.push("/dashboard");
    router.refresh();
  };

  return (
    <div className="max-w-md mx-auto mt-16 p-8">
      {/* Progress bar */}
      <div className="flex items-center gap-2 mb-8">
        {STEPS.map((step, i) => (
          <div key={step} className="flex items-center gap-2 flex-1">
            <div className={`w-8 h-8 rounded-full flex items-center justify-center text-sm font-bold transition-colors ${
              i <= currentStep ? "bg-black text-white" : "bg-gray-100 text-gray-400"
            }`}>
              {i + 1}
            </div>
            {i < STEPS.length - 1 && (
              <div className={`h-0.5 flex-1 transition-colors ${i < currentStep ? "bg-black" : "bg-gray-200"}`} />
            )}
          </div>
        ))}
      </div>

      {/* Step 0 — Welcome */}
      {currentStep === 0 && (
        <div className="text-center">
          <h2 className="text-2xl font-bold mb-2">Welcome to MyApp!</h2>
          <p className="text-gray-500 mb-6">Let's get you set up in under a minute.</p>
          <button
            onClick={() => setCurrentStep(1)}
            className="bg-black text-white px-6 py-2.5 rounded-lg font-medium"
          >
            Get Started
          </button>
        </div>
      )}

      {/* Step 1 — Role */}
      {currentStep === 1 && (
        <div>
          <h2 className="text-xl font-bold mb-2">What's your role?</h2>
          <p className="text-gray-500 mb-6">This helps us personalize your experience.</p>
          <div className="grid grid-cols-2 gap-3 mb-6">
            {["Developer", "Designer", "Product Manager", "Founder"].map((r) => (
              <button
                key={r}
                onClick={() => setRole(r)}
                className={`p-4 border-2 rounded-lg text-left transition-colors ${
                  role === r ? "border-black bg-gray-50" : "border-gray-200 hover:border-gray-400"
                }`}
              >
                <p className="font-medium text-sm">{r}</p>
              </button>
            ))}
          </div>
          <button
            onClick={handleRoleNext}
            disabled={!role || isLoading}
            className="w-full bg-black text-white py-2.5 rounded-lg font-medium disabled:opacity-50"
          >
            {isLoading ? "Saving..." : "Continue"}
          </button>
        </div>
      )}

      {/* Step 2 — Team */}
      {currentStep === 2 && (
        <div>
          <h2 className="text-xl font-bold mb-2">Team size?</h2>
          <p className="text-gray-500 mb-6">Just you, or a whole crew?</p>
          <div className="space-y-2 mb-6">
            {["Just me", "2–5", "6–20", "20+"].map((size) => (
              <button
                key={size}
                onClick={() => setTeamSize(size)}
                className={`w-full p-3 border-2 rounded-lg text-left text-sm transition-colors ${
                  teamSize === size ? "border-black bg-gray-50" : "border-gray-200"
                }`}
              >
                {size} {size === "Just me" ? "person" : "people"}
              </button>
            ))}
          </div>
          <button
            onClick={handleTeamNext}
            disabled={!teamSize || isLoading}
            className="w-full bg-black text-white py-2.5 rounded-lg font-medium disabled:opacity-50"
          >
            {isLoading ? "Saving..." : "Continue"}
          </button>
        </div>
      )}

      {/* Step 3 — Done */}
      {currentStep === 3 && (
        <div className="text-center">
          <div className="text-5xl mb-4">🎉</div>
          <h2 className="text-2xl font-bold mb-2">You're all set!</h2>
          <p className="text-gray-500 mb-6">Welcome aboard. Let's build something great.</p>
          <button
            onClick={handleComplete}
            disabled={isLoading}
            className="bg-black text-white px-6 py-2.5 rounded-lg font-medium disabled:opacity-50"
          >
            {isLoading ? "Loading..." : "Go to Dashboard →"}
          </button>
        </div>
      )}

      {error && (
        <p className="mt-4 text-sm text-red-600 text-center">{error.message}</p>
      )}
    </div>
  );
}
```

## 🚪 Sign Out

Use `useSignIn` hook's return context does not include sign-out. Import `signOut` as a standalone function, or use the `useAuth` extension:

```tsx
"use client";

import { signOut } from "@bolkauth/react";
import { useRouter } from "next/navigation";

export function SignOutButton() {
  const router = useRouter();

  const handleSignOut = async () => {
    await signOut();
    router.push("/sign-in");
    router.refresh();
  };

  return (
    <button
      onClick={handleSignOut}
      className="text-sm text-red-600 hover:underline"
    >
      Sign Out
    </button>
  );
}
```

## ♻️ Hook Dependency Summary

| Hook | Network Requests | When to Use |
|---|---|---|
| `useAuth()` | None — reads context | Conditional UI (signed in / out) |
| `useUser()` | GET `/api/auth/user` | Displaying user data, avatar, profile |
| `useSignIn()` | POST `/api/auth/sign-in/email` | Sign-in form |
| `useSignUp()` | POST `/api/auth/sign-up` | Registration form |
| `useOAuth()` | GET `/api/auth/sign-in/oauth/:provider` | Social login buttons |
| `useOnboarding()` | POST `/api/auth/onboarding/*` | Multi-step onboarding wizard |
