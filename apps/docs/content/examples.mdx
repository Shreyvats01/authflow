## 3. User Profile Page with Edit

A profile page that fetches user data server-side and allows editing the name via a Server Action:

```tsx
// app/settings/profile/page.tsx
import { createServerHelpers } from "@bolkauth/nextjs";

const { requireAuth } = createServerHelpers(auth);

  const user = await requireAuth();

  return (
```

```ts
// app/settings/profile/actions.ts
"use server";

import { createServerHelpers } from "@bolkauth/nextjs";
import { auth } from "@/lib/auth";
import { revalidatePath } from "next/cache";
import { redirect } from "next/navigation";

const { requireAuth } = createServerHelpers(auth);

export async function updateName(formData: FormData) {
  const user = await requireAuth();

  const name = (formData.get("name") as string).trim();
  if (name.length < 2) return;

  await fetch(`${process.env.NEXT_PUBLIC_APP_URL}/api/auth/user`, {
    method: "PATCH",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ name }),
  });

  revalidatePath("/settings/profile");
  redirect("/settings/profile?saved=1");
}
```

## 4. Sign-Out Button

```tsx
// components/auth/sign-out-button.tsx
"use client";

import { signOut } from "@bolkauth/react";
import { useRouter } from "next/navigation";
import { useState } from "react";

export function SignOutButton({ className }: { className?: string }) {
  const router = useRouter();
  const [isLoading, setIsLoading] = useState(false);

  const handleSignOut = async () => {
    setIsLoading(true);
    await signOut();
    router.push("/sign-in");
    router.refresh();
  };

  return (
    <button
      onClick={handleSignOut}
      disabled={isLoading}
      className={className ?? "text-sm text-red-600 hover:underline disabled:opacity-50"}
    >
      {isLoading ? "Signing out..." : "Sign Out"}
    </button>
  );
}
```

## 5. Protected API Route

Create an API route that reads the authenticated user's session before processing:

```ts
// app/api/projects/route.ts
import { createServerHelpers } from "@bolkauth/nextjs";
import { auth } from "@/lib/auth";
import { NextResponse } from "next/server";

const { getUser } = createServerHelpers(auth);

export async function GET() {
  const user = await getUser();

  if (!user) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  // Fetch user-specific data
  const projects = await db.query.projects.findMany({
    where: eq(projects.userId, user.id),
    orderBy: desc(projects.createdAt),
  });

  return NextResponse.json({ projects });
}

export async function POST(request: Request) {
  const user = await getUser();
  if (!user) {
    return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
  }

  const body = await request.json();
  const { name, description } = body;

  if (!name?.trim()) {
    return NextResponse.json({ error: "Project name is required" }, { status: 422 });
  }

  const project = await db.insert(projectsTable).values({
    id: crypto.randomUUID(),
    userId: user.id,
    name: name.trim(),
    description: description?.trim() ?? null,
  }).returning();

  return NextResponse.json({ project: project[0] }, { status: 201 });
}
```

## 6. Full Auth Page (Sign-In + Sign-Up Toggle)

A single page that switches between sign-in and sign-up forms:

```tsx
// app/auth/page.tsx
"use client";

import { useSignIn, useSignUp, useOAuth } from "@bolkauth/react";
import { useRouter } from "next/navigation";
import { useState } from "react";

export default function AuthPage() {
  const router = useRouter();
  const [mode, setMode] = useState<"signin" | "signup">("signin");
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");

  const { signIn, isLoading: signingIn, error: signInError, reset: resetSignIn } = useSignIn();
  const { signUp, isLoading: signingUp, error: signUpError, reset: resetSignUp } = useSignUp();
  const { signInWithOAuth, isLoading: oauthLoading } = useOAuth();

  const isLoading = signingIn || signingUp || oauthLoading;
  const error = mode === "signin" ? signInError : signUpError;

  const handleModeSwitch = (newMode: "signin" | "signup") => {
    setMode(newMode);
    resetSignIn();
    resetSignUp();
    setEmail("");
    setPassword("");
    setName("");
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (mode === "signin") {
      const result = await signIn({ email, password });
      if (!result.error) router.push("/dashboard");
    } else {
      const result = await signUp({ email, password, name });
      if (!result.error) router.push("/onboarding");
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-gray-50 to-gray-100">
      <div className="w-full max-w-sm bg-white rounded-2xl shadow-lg border p-8">
        {/* Tab toggle */}
        <div className="flex rounded-lg bg-gray-100 p-1 mb-6">
          {(["signin", "signup"] as const).map((m) => (
            <button
              key={m}
              onClick={() => handleModeSwitch(m)}
              className={`flex-1 py-2 rounded-md text-sm font-medium transition-all ${
                mode === m
                  ? "bg-white shadow-sm text-black"
                  : "text-gray-500 hover:text-gray-700"
              }`}
            >
              {m === "signin" ? "Sign In" : "Sign Up"}
            </button>
          ))}
        </div>

        {/* Social */}
        <div className="space-y-2 mb-5">
          <button
            onClick={() => signInWithOAuth("github")}
            disabled={isLoading}
            className="w-full flex items-center justify-center gap-2 py-2 border rounded-lg text-sm hover:bg-gray-50 disabled:opacity-50"
          >
            <span>GitHub</span>
          </button>
          <button
            onClick={() => signInWithOAuth("google")}
            disabled={isLoading}
            className="w-full flex items-center justify-center gap-2 py-2 border rounded-lg text-sm hover:bg-gray-50 disabled:opacity-50"
          >
            <span>Google</span>
          </button>
        </div>

        <div className="relative my-4">
          <div className="absolute inset-0 flex items-center">
            <div className="w-full border-t" />
          </div>
          <div className="relative flex justify-center text-xs">
            <span className="bg-white px-2 text-gray-400">or</span>
          </div>
        </div>

        {error && (
          <div className="mb-3 p-2.5 bg-red-50 border border-red-100 rounded-lg text-xs text-red-600">
            {error.message}
          </div>
        )}

        <form onSubmit={handleSubmit} className="space-y-3">
          {mode === "signup" && (
            <input
              type="text"
              placeholder="Full name"
              value={name}
              onChange={(e) => setName(e.target.value)}
              className="w-full px-3 py-2.5 border rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-black"
            />
          )}
          <input
            type="email"
            placeholder="Email"
            value={email}
            onChange={(e) => { setEmail(e.target.value); resetSignIn(); resetSignUp(); }}
            className="w-full px-3 py-2.5 border rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-black"
            required
          />
          <input
            type="password"
            placeholder="Password"
            value={password}
            onChange={(e) => { setPassword(e.target.value); resetSignIn(); resetSignUp(); }}
            className="w-full px-3 py-2.5 border rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-black"
            required
            minLength={8}
          />
          <button
            type="submit"
            disabled={isLoading}
            className="w-full bg-black text-white py-2.5 rounded-lg text-sm font-medium disabled:opacity-50"
          >
            {isLoading
              ? "Please wait..."
              : mode === "signin"
              ? "Sign In"
              : "Create Account"}
          </button>
        </form>
      </div>
    </div>
  );
}
```

## 8. Conditional Page Guard (Client-Side)

For pure client components that need to gate access:

```tsx
"use client";

import { useAuth } from "@bolkauth/react";
import { useRouter } from "next/navigation";
import { useEffect } from "react";

export function withAuth<T extends object>(Component: React.ComponentType<T>) {
  return function ProtectedComponent(props: T) {
    const router = useRouter();
    const { isLoaded, isSignedIn } = useAuth();

    useEffect(() => {
      if (isLoaded && !isSignedIn) {
        router.push("/sign-in");
      }
    }, [isLoaded, isSignedIn, router]);

    if (!isLoaded) {
      return (
        <div className="min-h-screen flex items-center justify-center">
          <div className="w-6 h-6 border-2 border-black border-t-transparent rounded-full animate-spin" />
        </div>
      );
    }

    if (!isSignedIn) return null;

    return <Component {...props} />;
  };
}

// const ProtectedDashboard = withAuth(DashboardComponent);
```
